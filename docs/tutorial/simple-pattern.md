# Simple Pattern Tutorial

Learn how to use `SQLAlchemyRepository` for straightforward database operations without separate DTOs.

## When to Use

The simple pattern is perfect when:

- Your model is your schema (no separate validation models)
- You want minimal boilerplate
- You're building prototypes or internal tools
- Your create and update operations use the same fields

## Setup

### 1. Define Your Model

```python
from sqlalchemy import String, create_engine
from sqlalchemy.orm import Mapped, mapped_column, Session
from database_utils.sqlalchemy import SQLAlchemyBase

class User(SQLAlchemyBase):
    __tablename__ = "users"

    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(100))
    email: Mapped[str] = mapped_column(String(100), unique=True)
```

### 2. Create Your Repository

```python
from database_utils.sqlalchemy import SQLAlchemyRepository

class UserRepository(SQLAlchemyRepository):
    model_type = User
```

### 3. Initialize Database

```python
engine = create_engine("sqlite:///db.sqlite")
SQLAlchemyBase.metadata.create_all(engine)
```

## Operations

### Create

Create a new record with `save()`:

```python
with Session(engine) as session:
    repo = UserRepository(session)
    
    # Create a new instance
    user = User(name="Alice", email="alice@example.com")
    
    # Save it (detects it's not in session, adds it)
    saved = repo.save(user)
    
    print(saved.id)  # ID auto-generated by database
    print(saved.name)  # "Alice"
```

### Read

Retrieve records with `list()` and `get()`:

```python
with Session(engine) as session:
    repo = UserRepository(session)
    
    # Get all users
    all_users = repo.list()
    print(len(all_users))
    
    # Get by ID
    user = repo.get(1)
    if user:
        print(user.name)
```

### Update

Modify and save with `save()`:

```python
with Session(engine) as session:
    repo = UserRepository(session)
    
    # Retrieve
    user = repo.get(1)
    
    # Modify
    user.name = "Alice Updated"
    
    # Save (detects it's already tracked, just flushes)
    updated = repo.save(user)
    
    print(updated.name)  # "Alice Updated"
```

### Delete

Remove records with `delete()`:

```python
with Session(engine) as session:
    repo = UserRepository(session)
    
    # Retrieve
    user = repo.get(1)
    
    # Delete
    repo.delete(user)
    
    # After session commit, it's removed from database
```

## Complete Example

```python
from sqlalchemy import String, create_engine
from sqlalchemy.orm import Mapped, mapped_column, Session
from database_utils.sqlalchemy import SQLAlchemyBase, SQLAlchemyRepository

# 1. Define model
class User(SQLAlchemyBase):
    __tablename__ = "users"
    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(100))
    email: Mapped[str] = mapped_column(String(100), unique=True)

# 2. Create repository
class UserRepository(SQLAlchemyRepository):
    model_type = User

# 3. Setup database
engine = create_engine("sqlite:///db.sqlite")
SQLAlchemyBase.metadata.create_all(engine)

# 4. Use it
with Session(engine) as session:
    repo = UserRepository(session)
    
    # Create
    user = User(name="Alice", email="alice@example.com")
    saved = repo.save(user)
    
    # Read
    user = repo.get(saved.id)
    print(user.name)  # Alice
    
    # Update
    user.name = "Alice Updated"
    repo.save(user)
    
    # Delete
    repo.delete(user)
    
    # Verify deletion
    deleted = repo.get(saved.id)
    print(deleted)  # None (after commit)
```

## Tips

### Autocommit

The repository calls `flush()` to make changes visible within the session, but you still need to commit:

```python
with Session(engine) as session:
    repo = UserRepository(session)
    user = User(name="Alice", email="alice@example.com")
    saved = repo.save(user)
    # Changes are visible in this session
    
    # But not persisted to DB until:
    session.commit()
```

### Working with Mixins

Combine with model mixins for extra functionality:

```python
from database_utils.sqlalchemy.mixins import TimestampMixin

class User(SQLAlchemyBase, TimestampMixin):
    __tablename__ = "users"
    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(100))
    email: Mapped[str] = mapped_column(String(100), unique=True)
    # created_at and updated_at automatically included
```

### Custom Queries

Extend your repository with custom methods:

```python
class UserRepository(SQLAlchemyRepository):
    model_type = User
    
    def find_by_email(self, email: str) -> User | None:
        stmt = select(self.model_type).where(self.model_type.email == email)
        return self.session.scalar(stmt)

# Usage
user = repo.find_by_email("alice@example.com")
```
